import requests
import json

url = "https://api.example.test/users"

payload = json.dumps({
  "name": "Ana",
  "tags": [
    "a",
    "b"
  ]
})
headers = {
  'Content-Type': 'application/json'
}

response = requests.request("POST", url, headers=headers, data=payload, allow_redirects=False, timeout=5)

print(response.text)

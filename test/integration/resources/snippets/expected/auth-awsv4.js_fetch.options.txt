const controller = new AbortController();
const timerId = setTimeout(() => controller.abort(), 5000);
const myHeaders = new Headers();
myHeaders.append("X-Amz-Date", "20150830T123600Z");
myHeaders.append("Authorization", "AWS4-HMAC-SHA256 Credential=AKIDEXAMPLE/20150830/us-east-1/execute-api/aws4_request, SignedHeaders=host;x-amz-date, Signature=0123456789abcdef");

const requestOptions = {
    method: "GET",
    headers: myHeaders,
    signal: controller.signal,
    redirect: "manual"
};

fetch("https://abc123.execute-api.us-east-1.amazonaws.com/prod/items", requestOptions)
    .then((response) => response.text())
    .then((result) => console.log(result))
    .catch((error) => console.error(error))
    .finally(() => clearTimeout(timerId));

const controller = new AbortController();
const timerId = setTimeout(() => controller.abort(), 5000);
const myHeaders = new Headers();
myHeaders.append("Authorization", "Digest username=\"ana\", realm=\"xenon@test\", nonce=\"n1\", uri=\"/digest\", algorithm=MD5, response=\"6629fae49393a05397450978507c4ef1\", qop=auth, nc=00000001, cnonce=\"0a4f113b\"");

const requestOptions = {
    method: "GET",
    headers: myHeaders,
    signal: controller.signal,
    redirect: "manual"
};

fetch("https://api.example.test/digest", requestOptions)
    .then((response) => response.text())
    .then((result) => console.log(result))
    .catch((error) => console.error(error))
    .finally(() => clearTimeout(timerId));

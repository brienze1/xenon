# Postman v3 fixtures (XEN-009)

All values are fake. Fixture secrets are distinct `s3cr3t-fixture-*` strings so the scenarios can prove that none of
them reaches a workspace file, a log line or an error.

`rich.json` generator (`python3 -I v21_to_v3_json.py rich-v2.1.json rich.json`):

```python
import json, sys

LANGS = {"json", "text", "xml", "html", "javascript"}


def rows(items, keep=("key", "value", "disabled", "description")):
    return [{k: r[k] for k in keep if k in r} for r in items or []]


def scripts(events, prefix):
    out = []
    for e in events or []:
        kind = {"prerequest": "beforeRequest", "test": "afterResponse"}[e["listen"]]
        out.append({"type": prefix + kind, "code": "\n".join(e["script"]["exec"]), "language": "text/javascript"})
    return out


def auth(a, as_list):
    if a is None:
        return None
    if a["type"] == "noauth":
        return [{"id": "auth-noauth", "name": "noauth", "type": "noauth"}] if as_list else {"type": "noauth"}
    creds = [{"key": r["key"], "value": r["value"]} for r in a.get(a["type"], [])]
    if as_list:
        return [{"id": "auth-" + a["type"], "name": a["type"] + " auth", "type": a["type"], "credentials": creds}]
    return {"type": a["type"], "credentials": creds}


def body(b):
    if not b:
        return None
    mode = b["mode"]
    if mode == "raw":
        lang = b.get("options", {}).get("raw", {}).get("language", "text")
        return {"type": lang if lang in LANGS else "text", "content": b["raw"]}
    if mode == "formdata":
        return {"type": "formdata", "content": [{k: r[k] for k in ("key", "type", "value", "src", "disabled", "contentType", "description") if k in r} for r in b["formdata"]]}
    if mode == "urlencoded":
        return {"type": "urlencoded", "content": rows(b["urlencoded"])}
    if mode == "graphql":
        return {"type": "graphql", "content": dict(b["graphql"])}
    if mode == "file":
        return {"type": "file", "content": {"src": b["file"]["src"]}}
    raise ValueError(mode)


def example(r):
    o = r["originalRequest"]
    ex = {"$kind": "http-example", "id": r["id"], "name": r["name"],
          "request": {"url": o["url"]["raw"], "method": o["method"], "headers": rows(o.get("header"))},
          "response": {"statusCode": r["code"], "statusText": r["status"], "headers": rows(r.get("header"), ("key", "value")),
                       "body": {"type": r.get("_postman_previewlanguage", "text"), "content": r.get("body", "")}}}
    return ex


def item(i):
    if "item" in i:
        f = {"$kind": "collection", "id": i["id"], "name": i["name"]}
        if "description" in i:
            f["description"] = i["description"]
        if "auth" in i:
            f["auth"] = auth(i["auth"], True)
        f["scripts"] = scripts(i.get("event"), "http:")
        f["items"] = [item(c) for c in i["item"]]
        return f
    r = i["request"]
    q = {"$kind": "http-request", "id": i["id"], "name": i["name"], "url": r["url"]["raw"], "method": r["method"],
         "headers": rows(r.get("header")), "queryParams": rows(r["url"].get("query")),
         "pathVariables": rows(r["url"].get("variable"), ("key", "value", "description"))}
    if body(r.get("body")):
        q["body"] = body(r.get("body"))
    if "auth" in r:
        q["auth"] = auth(r["auth"], False)
    q["scripts"] = scripts(i.get("event"), "")
    q["settings"] = dict(i["protocolProfileBehavior"]) if "protocolProfileBehavior" in i else {"disabledSystemHeaders": []}
    if "description" in r:
        q["description"] = r["description"]
    q["examples"] = [example(x) for x in i.get("response", [])]
    return q


def main(src, dst):
    c = json.load(open(src))
    out = {"id": c["info"]["_postman_id"], "name": c["info"]["name"], "$kind": "collection",
           "description": c["info"]["description"],
           "variables": [dict({"key": v["key"], "value": v["value"], "disabled": v.get("disabled", False), "secret": False}, **({"description": v["description"]} if "description" in v else {})) for v in c.get("variable", [])],
           "auth": auth(c.get("auth"), True), "scripts": scripts(c.get("event"), "http:"),
           "items": [item(i) for i in c["item"]]}
    json.dump(out, open(dst, "w"), indent=2)
    open(dst, "a").write("\n")


main(sys.argv[1], sys.argv[2])
```
